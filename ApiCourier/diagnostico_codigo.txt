using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using ApiCourier.Data;
using ApiCourier.Models;

namespace ApiCourier.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class ClientesController : ControllerBase
    {
        private readonly AppDbContext _context;

        public ClientesController(AppDbContext context)
        {
            _context = context;
        }

        // GET: api/clientes?buscar=123
        [HttpGet]
        public async Task<IActionResult> Get([FromQuery] string? buscar)
        {
            var query = _context.Clientes.Where(c => c.Estado);

            if (!string.IsNullOrWhiteSpace(buscar))
            {
                var filtro = buscar.Trim().ToLower();
                query = query.Where(c =>
                    c.NumeroDocumento.ToLower().Contains(filtro) ||
                    c.NombreRazonSocial.ToLower().Contains(filtro)
                );
            }

            var resultado = await query.ToListAsync();
            return Ok(resultado);
        }

        // POST: api/clientes (Crear o Actualizar)
        [HttpPost]
        public async Task<IActionResult> Post([FromBody] Cliente dto)
        {
            if (dto == null || string.IsNullOrWhiteSpace(dto.NumeroDocumento))
                return BadRequest("Datos de cliente inválidos.");

using Microsoft.EntityFrameworkCore;
using ApiCourier.Data;
using Npgsql;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();

// Obtener la cadena de conexión (prioriza variable de entorno de Render, luego appsettings.json)
var rawConnectionString = Environment.GetEnvironmentVariable("DB_CONNECTION_STRING") 
    ?? builder.Configuration.GetConnectionString("DefaultConnection");

// Si viene en formato postgresql:// (tipo URL), la convertimos automáticamente al formato Npgsql
string finalConnectionString = rawConnectionString;
if (rawConnectionString != null && rawConnectionString.StartsWith("postgresql://"))
{
    var uri = new Uri(rawConnectionString);
    var userInfo = uri.UserInfo.Split(':');
    var builderConn = new NpgsqlConnectionStringBuilder
    {
        Host = uri.Host,
        Port = uri.Port > 0 ? uri.Port : 5432,
        Username = userInfo[0],
        Password = userInfo.Length > 1 ? userInfo[1] : "",
        Database = uri.AbsolutePath.TrimStart('/'),
        SslMode = SslMode.Require,
        TrustServerCertificate = true
    };
    finalConnectionString = builderConn.ToString();
}

builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseNpgsql(finalConnectionString));

builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowAll", policy =>
    {
        policy.AllowAnyOrigin().AllowAnyMethod().AllowAnyHeader();
    });
});

var app = builder.Build();

app.UseCors("AllowAll");
app.UseAuthorization();
app.MapControllers();

app.Run();
